1use serde::Deserialize;
2use std::f64::consts::PI;
3use std::fs::File;
4use std::io::BufWriter;
5use std::path::Path;
6
7const TILE_SIZE: f64 = 256.0;
8const BAND_HEIGHT: u32 = 512; // Render 512 pixels high at a time
9
10#[derive(Deserialize, Debug)]
11struct LatLng {
12    lat: f64,
13    lng: f64,
14}
15
16// Global pixel coordinate
17#[derive(Debug, Clone, Copy)]
18struct Point {
19    x: f64,
20    y: f64,
21}
22
23fn project(lat: f64, lng: f64, zoom: f64) -> Point {
24    let lat_rad = lat * PI / 180.0;
25    let n = f64::powf(2.0, zoom);
26    let x = (lng + 180.0) / 360.0 * n * TILE_SIZE;
27    let y = (1.0 - ((lat_rad.tan() + (1.0 / lat_rad.cos())).ln() / PI)) / 2.0 * n * TILE_SIZE;
28    Point { x, y }
29}

31pub async fn render_map(
32    output_path: String,
33    trace_json: String,
34    theme: String,
35    width: u32,
36    height: u32,
37    center_lat: f64,
38    center_lng: f64,
39    zoom: f64,
40) -> i32 {
41    // 1. Parse GPS trace
42    let trace: Vec<LatLng> = serde_json::from_str(&trace_json).unwrap_or_default();
43
44    // 2. Global bounds of the entire poster
45    let center_pt = project(center_lat, center_lng, zoom);
46    let poster_left = center_pt.x - (width as f64 / 2.0);
47    let poster_top = center_pt.y - (height as f64 / 2.0);
48
49    // 3. Setup PNG encoder
50    let file = match File::create(Path::new(&output_path)) {
51        Ok(f) => f,
52        Err(e) => {
53            println!("Failed to create output file: {}", e);
54            return 2;
55        }
56    };
57    let ref mut w = BufWriter::new(file);
58    let mut encoder = png::Encoder::new(w, width, height);
59    encoder.set_color(png::ColorType::Rgba);
60    encoder.set_depth(png::BitDepth::Eight);
61    let mut writer = match encoder.write_header() {
62        Ok(w) => w,
63        Err(_) => return 3,
64    };
65
66    let client = reqwest::Client::new();
67    let map_style = if theme.to_lowercase() == "dark" { "8" } else { "7" };
68
69    // 4. Render Horizontal Bands
70    for band_y in (0..height).step_by(BAND_HEIGHT as usize) {
71        let current_band_height = if band_y + BAND_HEIGHT > height {
72            height - band_y
73        } else {
74            BAND_HEIGHT
75        };
76
77        let mut pixmap = match tiny_skia::Pixmap::new(width, current_band_height) {
78            Some(p) => p,
79            None => return 4,
80        };
81
82        let band_top = poster_top + band_y as f64;
83        let band_bottom = band_top + current_band_height as f64;
84
85        // Find intersecting tiles
86        let tile_start_x = (poster_left / TILE_SIZE).floor() as i64;
87        let tile_end_x = ((poster_left + width as f64) / TILE_SIZE).floor() as i64;
88        let tile_start_y = (band_top / TILE_SIZE).floor() as i64;
89        let tile_end_y = (band_bottom / TILE_SIZE).floor() as i64;
90
91        // Download and draw tiles
92        for tx in tile_start_x..=tile_end_x {
93            for ty in tile_start_y..=tile_end_y {
94                let url = format!(
95                    "https://wprd01.is.autonavi.com/appmaptile?lang=zh_cn&size=1&style={}&x={}&y={}&z={}",
96                    map_style, tx, ty, zoom as i32
97                );
98
99                if let Ok(resp) = client.get(&url).send().await {
100                    if let Ok(bytes) = resp.bytes().await {
101                        // Decode PNG tile (Amap returns PNGs for static maps/tiles)
102                        if let Ok(tile_img) = tiny_skia::Pixmap::decode_png(&bytes) {
103                            let draw_x = (tx as f64 * TILE_SIZE) - poster_left;
104                            let draw_y = (ty as f64 * TILE_SIZE) - band_top;
105
106                            pixmap.draw_pixmap(
107                                draw_x as i32,
108                                draw_y as i32,
109                                tile_img.as_ref(),
110                                &tiny_skia::PixmapPaint::default(),
111                                tiny_skia::Transform::identity(),
112                                None,
113                            );
114                        }
115                    }
116                }
117            }
118        }
119
120        // Draw path segments intersecting this band
121        if trace.len() > 1 {
122            let mut pb = tiny_skia::PathBuilder::new();
123            let mut first = true;
124
125            for point in &trace {
126                let pt = project(point.lat, point.lng, zoom);
127                let local_x = (pt.x - poster_left) as f32;
128                let local_y = (pt.y - band_top) as f32; // Negative if above band, positive if below
129
130                if first {
131                    pb.move_to(local_x, local_y);
132                    first = false;
133                } else {
134                    pb.line_to(local_x, local_y);
135                }
136            }
137
138            if let Some(path) = pb.finish() {
139                let mut paint = tiny_skia::Paint::default();
140                paint.set_color_rgba8(255, 69, 58, 255); // Red path
141                paint.anti_alias = true;
142                
143                let mut stroke = tiny_skia::Stroke::default();
144                stroke.width = 12.0; // Path width
145                stroke.line_cap = tiny_skia::LineCap::Round;
146                stroke.line_join = tiny_skia::LineJoin::Round;
147
148                pixmap.stroke_path(&path, &paint, &stroke, tiny_skia::Transform::identity(), None);
149            }
150        }
151
152        // 5. Stream the chunk bytes to PNG encoder row by row
153        let bytes = pixmap.data();
154        if let Err(e) = writer.write_image_data(bytes) {
155            println!("Failed to write chunk: {}", e);
156            return 5;
157        }
158    }
159
160    0
161}
